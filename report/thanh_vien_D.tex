\section{Thuật toán GRASP, Thực nghiệm và Phân tích Thống kê}

\subsection{Thuật toán GRASP}
GRASP (Greedy Randomized Adaptive Search Procedure; Feo và Resende, 1995) là metaheuristic lặp hai pha gồm \emph{xây dựng nghiệm tham lam ngẫu nhiên} (RCL) và \emph{tìm kiếm cục bộ} (Local Search).
\begin{enumerate}
 \item \textbf{Pha xây dựng (RCL Chvátal):} Tại mỗi bước, tính tỉ số $s_j = c_j / g_j$ với $g_j$ là số nút chưa phủ mà camera $j$ phủ được. Gọi $s_{\min} = \min_j s_j$ và $s_{\max} = \max_j s_j$, danh sách ứng viên hạn chế là
 \[
  \mathrm{RCL} = \{ j : s_j \le s_{\min} + \alpha (s_{\max} - s_{\min}) \}.
 \]
 Chọn ngẫu nhiên một camera trong $\mathrm{RCL}$ cho tới khi mọi nút đều được phủ, sau đó xóa các camera dư thừa theo thứ tự chi phí giảm dần. Ở vòng lặp đầu tiên đặt $\alpha = 0$ để thu ngay nghiệm tham lam chuẩn; các vòng tiếp theo chọn $\alpha \sim \mathcal{U}(0.02, 0.12)$.
 \item \textbf{Pha tìm kiếm cục bộ:} Kết hợp hai lân cận: (i) \emph{hoán đổi 1--1} thay camera $j$ đang chọn bằng camera $k$ chưa chọn rẻ hơn phủ toàn bộ các nút chỉ do $j$ phủ; (ii) \emph{phá hủy 1 -- sửa chữa tham lam} (destroy-1 \& repair) trên tối đa 30 camera có chi phí cao nhất, kèm xóa cột dư thừa.
\end{enumerate}

\subsection{Thiết kế thực nghiệm chung (10 thể hiện, 3 mức kích thước)}
\begin{itemize}
 \item \textbf{Tập dữ liệu:} 10 thể hiện thuộc 3 mức kích thước: 4 thể hiện nhỏ tự sinh ($m \in [10, 25], n \in [20, 30]$), 3 thể hiện vừa tự sinh ($m \in [100, 200], n \in [400, 1000]$) và 3 thể hiện chuẩn OR-Library (\texttt{scp41}, \texttt{scp51}, \texttt{scp61} với tối ưu công bố lần lượt $429$, $253$, $138$).
 \item \textbf{Thuật toán đối sánh:} \texttt{EXACT} (chỉ chạy cho $n \le 30$), \texttt{GREEDY}, \texttt{HC\_FI}, \texttt{HC\_BI}, \texttt{TS}, \texttt{SA}, \texttt{GA}, \texttt{GA\_LS}, và \texttt{GRASP}.
 \item \textbf{Ngân sách và seed:} Mỗi thuật toán chạy độc lập 10 lần với \texttt{seed = 1..10}, ngân sách thời gian $0{,}5$ giây/lần chạy trên Python 3.12.9 (Windows 11). Toàn bộ 800 nghiệm đều đạt \texttt{HOP LE} qua \texttt{src/kiemtra.py}.
\end{itemize}

\subsection{Phân tích thống kê (Friedman và Wilcoxon)}
\begin{itemize}
 \item \textbf{Kiểm định Friedman} trên 10 thể hiện với 8 thuật toán (\texttt{GREEDY}, \texttt{HC\_FI}, \texttt{HC\_BI}, \texttt{TS}, \texttt{SA}, \texttt{GA}, \texttt{GA\_LS}, \texttt{GRASP}): $\chi^2_F = 19{,}9419$ ($\mathrm{df}=7$), $p = 0{,}0057 < 0{,}05$ $\Rightarrow$ \textbf{có khác biệt có ý nghĩa thống kê giữa các thuật toán}. Hạng trung bình Friedman (nhỏ hơn là tốt hơn): \texttt{TS} ($3{,}200$), \texttt{GA\_LS} ($3{,}850$), \texttt{HC\_BI} ($4{,}000$), \texttt{GRASP} ($4{,}050$), \texttt{HC\_FI} ($4{,}250$), \texttt{SA} ($4{,}900$), \texttt{GA} ($5{,}250$), \texttt{GREEDY} ($6{,}500$).
 \item \textbf{Kiểm định Wilcoxon signed-rank} ghép cặp theo từng $(\text{instance}, \text{seed})$ kèm hiệu chỉnh Holm ($\alpha = 0{,}05$): toàn bộ 7 thuật toán cải thiện (\texttt{TS}, \texttt{GA\_LS}, \texttt{GRASP}, \texttt{HC\_BI}, \texttt{HC\_FI}, \texttt{SA}, \texttt{GA}) đều vượt trội có ý nghĩa thống kê so với \texttt{GREEDY} ($p_{\mathrm{Holm}} \le 0{,}0047$). \texttt{TS} và \texttt{GA\_LS} vượt trội có ý nghĩa so với \texttt{SA} ($p_{\mathrm{Holm}} \le 0{,}027$), và \texttt{TS}, \texttt{GA\_LS} vượt trội so với \texttt{GA} ($p_{\mathrm{Holm}} \le 0{,}00067$). Giữa bộ ba dẫn đầu (\texttt{TS}, \texttt{GA\_LS}, \texttt{GRASP}) chưa có khác biệt có ý nghĩa thống kê ở ngân sách $0{,}5$ giây ($p_{\mathrm{Holm}} \ge 0{,}50$).
\end{itemize}

\subsection{Kết luận}
Các metaheuristic kết hợp tìm kiếm cục bộ (\texttt{TS}, \texttt{GA\_LS}, \texttt{GRASP}) đạt hiệu quả ổn định và vượt trội so với tham lam thuần túy và \texttt{GA} cơ bản: \texttt{GRASP} dẫn đầu trên \texttt{vua\_100x400\_s201} ($204{,}00$) và \texttt{scp41} (best $430$, gap $0{,}23\%$), \texttt{GA\_LS} chạm tối ưu $253$ trên \texttt{scp51} ($20\%$), còn \texttt{TS} có hạng trung bình Friedman tốt nhất ($3{,}200$) trên toàn bộ 10 thể hiện.
